const norm = (v: unknown) => String(v ?? "").normalize("NFD").replace(/\p{M}/gu, "").toLowerCase();
function filtrer(l: Ligne[], filtres: { champ: string; op: string; valeur: string }[]) {
  return l.filter((x) => filtres.every((f) => {
    const v = x[f.champ];
    const vs = Array.isArray(v) ? v.map(norm) : [norm(v)];
    if (f.op === "=") return vs.some((s) => s === norm(f.valeur));
    if (f.op === "contient") return vs.some((s) => s.includes(norm(f.valeur)));
    if (f.op === "!=") return vs.every((s) => s !== norm(f.valeur));
    return true;
  }));
}

// ─── (c) PRODUIRE : HTML libre → assaini (liste blanche, CSS en ligne borné) → rendu → OBSERVATION ───
const BALISES = new Set(["p", "br", "div", "span", "strong", "b", "em", "i", "u", "ul", "ol", "li", "table", "thead", "tbody", "tr", "th", "td", "h1", "h2", "h3", "hr", "a"]);
const CSS_PERMIS = new Set(["color", "background", "background-color", "border", "border-collapse", "padding", "margin", "font-weight", "font-size", "font-family", "text-align", "width"]);
function assainir(html: string): string {
  return html
    .replace(/<(script|style|iframe|object|embed|svg)\b[\s\S]*?<\/\1\s*>/gi, "")
    .replace(/<\/?([a-z0-9]+)([^>]*)>/gi, (tout, b: string, attrs: string) => {
      const balise = b.toLowerCase();
      if (!BALISES.has(balise)) return "";
      if (tout.startsWith("</")) return `</${balise}>`;
      const style = /style\s*=\s*"([^"]*)"/i.exec(attrs)?.[1] ?? "";
      const propre = style.split(";").map((d) => d.trim()).filter((d) => CSS_PERMIS.has(d.split(":")[0]!.trim().toLowerCase()) && !/url\(|expression/i.test(d)).join(";");
      const href = balise === "a" ? /href\s*=\s*"(https?:\/\/[^"]+|\/[^"]*)"/i.exec(attrs)?.[1] : undefined;
      return `<${balise}${propre ? ` style="${propre}"` : ""}${href ? ` href="${href}"` : ""}>`;
    });
}
/** Ce que la personne verra, décrit SANS le modèle : la boucle d'observation. */
function observerHtml(html: string) {
  const styles = [...html.matchAll(/<([a-z0-9]+) style="([^"]*)"/g)].map((m) => m[1]);
  const parBalise = styles.reduce<Record<string, number>>((a, b) => ((a[b] = (a[b] ?? 0) + 1), a), {});
  return {
    tableaux: (html.match(/<table/g) ?? []).length,
    lignesDeTableau: (html.match(/<tr/g) ?? []).length,
    paragraphes: (html.match(/<p[ >]/g) ?? []).length,
    elementsStyles: parBalise,
    paragraphesStyles: (html.match(/<p style=/g) ?? []).length,
    couleurs: [...new Set([...html.matchAll(/(#[0-9a-f]{3,6})/gi)].map((m) => m[1]!.toUpperCase()))],
    texte: html.replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim().slice(0, 400),
  };
}

async function versionCourante(nodeId: string) {
  const v = await prisma.fileVersion.findFirst({ where: { nodeId }, orderBy: { version: "desc" } });
  return v ? await getBlob(v.blobId) : null;
}

async function executer(nom: string, a: any): Promise<unknown> {
  if (nom === "lire") {
    const v = VUES[a.vue];
    if (!v) return { erreur: `vue inconnue ; vues : ${Object.keys(VUES).join(", ")}` };
    const inconnus = [...a.champs, ...a.filtres.map((f: any) => f.champ)].filter((c: string) => !v.champs.includes(c));
    if (inconnus.length) return { erreur: `champ(s) inconnu(s) : ${inconnus.join(", ")} ; champs de ${a.vue} : ${v.champs.join(", ")}` };
    const l = filtrer(await v.lire(), a.filtres).map((x) => Object.fromEntries(a.champs.map((c: string) => [c, x[c]])));
    JEUX.set(a.jeu, l);
    return { jeu: a.jeu, lignes: l.length, apercu: l.slice(0, 30) };
  }
  if (nom === "produire") {
    const fichier = path.join(OUT, a.nom.replace(/[\\/]/g, "-"));
    if (a.format === "xlsx") {
      const l = JEUX.get(a.jeu);
      if (!l) return { erreur: `jeu « ${a.jeu} » inconnu ; jeux : ${[...JEUX.keys()].join(", ")}` };
      const cols = [...Object.keys(l[0] ?? {}), ...a.colonnesAjoutees];
      const wb = new ExcelJS.Workbook(); const ws = wb.addWorksheet(a.nom.replace(/\.xlsx$/, "").slice(0, 31));
      ws.addRow(cols); for (const x of l) ws.addRow(cols.map((c) => { const v = x[c]; return Array.isArray(v) ? v.join(", ") : v ?? ""; }));
      await wb.xlsx.writeFile(fichier.endsWith(".xlsx") ? fichier : `${fichier}.xlsx`);
      return { ecrit: `Adam/Brouillons/${path.basename(fichier)}`, observation: { colonnes: cols, lignes: l.length, totaux: "aucun (aucune colonne additionnée sans demande)" } };
    }
    const propre = assainir(a.html);
    fs.writeFileSync(`${fichier}.html`, propre);
    return { ecrit: `Adam/Brouillons/${path.basename(fichier)}`, retireParAssainissement: a.html.length - propre.length, observation: observerHtml(propre) };
  }
  if (nom === "fichier") {
    const n = await prisma.driveNode.findFirst({ where: { ownerId: SA, type: "FILE", name: { contains: a.nomFichier.replace(/\.docx$/i, ""), mode: "insensitive" } } });
    if (!n) return { erreur: "fichier introuvable dans le Drive de la personne" };
    const octets = await versionCourante(n.id);
    const zip = await JSZip.loadAsync(octets!);
    let xml = await zip.file("word/document.xml")!.async("string");
    const compte: Record<string, number> = {};
    for (const r of a.remplacements) { const n0 = xml.split(r.avant).length - 1; compte[r.avant] = n0; xml = xml.split(r.avant).join(r.apres); }
    zip.file("word/document.xml", xml);
    const docx = path.join(OUT, n.name);
    fs.writeFileSync(docx, await zip.generateAsync({ type: "nodebuffer" }));
    let pdf: string | null = null;
    if (a.versFormat === "pdf") { execFileSync("soffice", ["--headless", "--convert-to", "pdf", "--outdir", OUT, docx], { stdio: "ignore", timeout: 120_000 }); pdf = docx.replace(/\.docx$/, ".pdf"); }
    const texte = xml.replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim();
    return { nouvelleVersion: `${n.name} v2 (v1 conservée : annulable)`, pdf: pdf ? `Adam/Brouillons/${path.basename(pdf)}` : null, observation: { remplacements: compte, texte: texte.slice(0, 400), pdfOctets: pdf ? fs.statSync(pdf).size : null } };
  }
  return { erreur: "outil inconnu" };
}

async function appeler(input: unknown[]) {
  const r = await fetch("https://api.openai.com/v1/responses", {
    method: "POST", headers: { "content-type": "application/json", authorization: "Bearer sk-proxy" },
    body: JSON.stringify({ model: MODELE, reasoning: { effort: "low" }, tools: OUTILS, input, instructions: "Tu es Adam, l'assistant de la Direction d'Adventum Pharma (couleur de la charte : #087084, texte blanc sur la couleur). Tu fais ce que la personne demande, avec tes outils génériques. Tu vérifies l'OBSERVATION rendue par un outil contre la demande avant de dire que c'est fait ; si elle ne correspond pas, tu corriges. Réponds en français, brièvement." }),
  });
  const j = await r.json() as any;
  if (j.error) throw new Error(JSON.stringify(j.error));
  coutUsd += (j.usage.input_tokens * 2 + j.usage.output_tokens * 12) / 1e6;
  return j;
}

async function jouer(id: string, tours: string[]) {
  const input: any[] = [];
  const journal: unknown[] = [];
  for (const t of tours) {
    input.push({ role: "user", content: t });
    for (let i = 0; i < 8; i++) {
      const j = await appeler(input);
      const appels = j.output.filter((o: any) => o.type === "function_call");
      input.push(...j.output);
      if (!appels.length) { journal.push({ demande: t, reponse: j.output_text ?? j.output.map((o: any) => o.content?.map((c: any) => c.text).join("")).join("") }); break; }
      for (const c of appels) {
        const args = JSON.parse(c.arguments);
        const res = await executer(c.name, args).catch((e) => ({ erreur: String(e) }));
        journal.push({ outil: c.name, args, res });
        input.push({ type: "function_call_output", call_id: c.call_id, output: JSON.stringify(res) });
      }
    }
  }
  return { id, journal };
}

async function main() {
  const conv: { id: string; tours: string[] }[] = JSON.parse(process.env.CONV!);
  const res = [];
  for (const c of conv) { res.push(await jouer(c.id, c.tours)); fs.writeFileSync(path.join(OUT, "maquette.json"), JSON.stringify({ coutUsd, res }, null, 2)); console.log(c.id, "coût cumulé", coutUsd.toFixed(4)); }
  process.exit(0);
}
